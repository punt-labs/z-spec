\documentclass{article}
\usepackage{fuzz}
\begin{document}

\section{Basic Types}

\begin{zed}
  [MSG, ADDR]
\end{zed}

\begin{zed}
  Flag ::= flSeen | flUnseen | flFlagged
\end{zed}

\section{Constants}

\begin{axdef}
  maxMsgs : \nat
\where
  maxMsgs = 100
\end{axdef}

\section{State}

\begin{schema}{Inbox}
  msgs : \power MSG \\
  flag : MSG \pfun Flag \\
  from : MSG \pfun ADDR
\where
  \dom flag = msgs \\
  \dom from = msgs \\
  \# msgs \leq maxMsgs
\end{schema}

\section{Operations}

\begin{schema}{Receive}
  \Delta Inbox \\
  m? : MSG \\
  sender? : ADDR
\where
  m? \notin msgs \\
  \# msgs < maxMsgs \\
  msgs' = msgs \cup \{m?\} \\
  flag' = flag \cup \{m? \mapsto flUnseen\} \\
  from' = from \cup \{m? \mapsto sender?\}
\end{schema}

\begin{schema}{Delete}
  \Delta Inbox \\
  m? : MSG
\where
  m? \in msgs \\
  msgs' = msgs \setminus \{m?\} \\
  flag' = \{m?\} \ndres flag \\
  from' = \{m?\} \ndres from
\end{schema}
